Find a Tutor

Search available tutors by subject and schedule.

View Generated SQL Query 
-- Dynamic query generated based on filters
SELECT 
    t.user_id, u.first_name, u.last_name,
    t.hourly_rate, t.meeting_mode,
    AVG(r.rating) AS avg_rating,
    COUNT(r.review_id) AS total_reviews
FROM TUTOR t
JOIN USER u ON t.user_id = u.user_id
JOIN TUTOR_SUBJECT ts ON t.user_id = ts.tutor_id
LEFT JOIN BOOKING b ON t.user_id = b.tutor_id
LEFT JOIN REVIEW r ON b.booking_id = r.booking_id
WHERE ts.subject_code = ?
GROUP BY t.user_id
ORDER BY avg_rating DESC;

My Bookings

Manage your tutoring session bookings.

View Generated SQL Query 
-- Query 3: Student's upcoming bookings
SELECT b.booking_id, s.subject_code,
       s.name AS subject_name,
       u_tutor.first_name, u_tutor.last_name,
       a.start_time, a.end_time,
       a.mode, a.location, b.booking_status
FROM BOOKING b
JOIN SUBJECT s ON b.subject_code = s.subject_code
JOIN USER u_tutor ON b.tutor_id = u_tutor.user_id
JOIN AVAILABILITY_SLOT a ON b.slot_id = a.slot_id
WHERE b.student_id = ?
  AND b.booking_status IN ('confirmed','pending')
ORDER BY a.start_time ASC;

Tutor Reviews

Aggregated ratings and student feedback.

View Generated SQL Query 
-- Query 2: Aggregate tutor reviews
SELECT t.user_id, u.first_name, u.last_name,
       COUNT(r.review_id) AS total_reviews,
       ROUND(AVG(r.rating), 2) AS average_rating
FROM TUTOR t
JOIN USER u ON t.user_id = u.user_id
LEFT JOIN BOOKING b ON t.user_id = b.tutor_id
LEFT JOIN REVIEW r ON b.booking_id = r.booking_id
GROUP BY t.user_id, u.first_name, u.last_name
ORDER BY average_rating DESC, total_reviews DESC;
